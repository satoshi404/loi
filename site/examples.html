Examples

Every file under examples/ is runnable with ./loi examples/NN.loi --run. The exit code is the program's return value. Every file declares a package; runtime helpers are reached through the std package (@inc std).

01_basics — recursion & control


package example_01

@inc std

factorial = (n: int): int
{
    if (n <= 1) { return 1; }
    return n * factorial(n - 1);
}

main =
{
    int x = 5;
    if (x < 10) { return factorial(x); }
    else { return 0; }
}
exit: 120

03_math — built-ins


package example_03

@inc std

main =
{
    double a = 2.0;
    double p = std.pow(a, 3.0);
    std.print_int(std.abs(-99));
    if (p == 8.0) { std.print_str("math ok"); }
    return 0;
}
prints 99, "math ok" ; exit 0

14_macros — object-like macro


package example_14

@inc std
@(define LIMIT 8)

scale = (n: int): int { return n * 2; }

main =
{
    int v = scale(LIMIT);
    std.print_int(v);
    return v;
}
prints 16 ; exit 16

25_fnpointers — typed function pointers


package example_25

@inc std
type CmpFn = (a: int, b: int) -> int;

my_less = (a: int, b: int): int { return a < b; }
apply = (f: (a:int,b:int)->int, x:int, y:int): int { return f(x, y); }

main = () -> int !IO
{
    std.print_int(apply(my_less, 2, 3));
    return 0;
}
exit: 0

26_macro_trio — stringify / paste / variadic


package example_26

@inc std
@(define STR(x) @stringify(x))
@(define CONCAT(a, b) @paste(a, b))
@(define LOG(level, ...msg) { @if(level >= 1) { std.print_str(msg); } })

int foobar = 77;

main = () -> int !IO
{
    std.print_str(STR(hello));        // "hello"
    int foo = 1, bar = 2;
    std.print_int(CONCAT(foo, bar));  // foobar -> 77
    LOG(1, "log: variadic works\n");
    return 0;
}
exit: 0
