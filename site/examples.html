Examples

Every file under examples/ is runnable with ./loi examples/NN.loi --run. The exit code is the program's return value. Every file declares an @package(name); runtime helpers are reached through the std package (@inc std).

01_basics — recursion & control


@package(examples_01_basics)

@inc std

factorial = (n: int): int
{
    if (n <= 1) { return 1; }
    return n * factorial(n - 1);
}

main =
{
    int x = 5;
    if (x < 10) { return factorial(x); }
    else { return 0; }
}
exit: 120

03_math — built-ins


@package(examples_03_math)

@inc std

main =
{
    double a = 2.0;
    double p = std.pow(a, 3.0);
    std.print_int(std.abs(-99));
    if (p == 8.0) { std.print_str("math ok"); }
    return 0;
}
prints 99, "math ok" ; exit 0

26_macro_trio — stringify / paste / variadic


@package(examples_26_macro_trio)

@inc std
@(define STR(x) @stringify(x))
@(define CONCAT(a, b) @paste(a, b))
@(define LOG(level, ...msg) { @if(level >= 1) { std.print_str(msg); } })

int foobar = 77;

main = () -> int !IO
{
    std.print_str(STR(hello));        // "hello"
    int foo = 1, bar = 2;
    std.print_int(CONCAT(foo, bar));  // foobar -> 77
    LOG(1, "log: variadic works\n");
    return 0;
}
exit: 0

27_effects — effects enforcement


@package(examples_27_effects)

@inc std

do_io = (): int !IO
{
    std.print_str("in do_io\n");
    return 42;
}

no_io = (): int { return 7; }

main = () -> int !IO
{
    int a = do_io();       // OK: caller has !IO

}

29_capabilities — linear permissions


@package(examples_29_capabilities)

@inc std

@capability(Read, Write)

peek = (buf: ptr, c: @cap(borrow Read)): int { return *buf; }
poke = (buf: ptr, c: @cap(Write)): void { *buf = 42; }

new_page = (buf: ptr): @cap(Write)
{
    @grant(Write) w;
    return w;
}

main = (): int !IO
{
    int p = 0;
    @cap(Write) w = new_page(&p);
    poke(&p, w);                      /* consumes w */
    @grant(Read) rd;
    @with_cap(Read) { peek(&p, rd); } /* ambient borrow, block-scoped */
    @drop(rd);
    region r(64) with Write { }       /* ambient cap released with region */
    return p;
}
exit: 42

tictactoe — interactive game (Playground)


@package(tictactoe)

@inc std

board: int[9] = {0, 0, 0, 0, 0, 0, 0, 0, 0};
current: int = 1; /* 1 = X, 2 = O */

print_board = (): void !IO
{
    std.print_str("\n");
    for (int i = 0; i < 9; i += 3) {
        char c0 = (board[i] == 1) ? 'X' : (board[i] == 2 ? 'O' : '0' + i + 1);
        char c1 = (board[i+1] == 1) ? 'X' : (board[i+1] == 2 ? 'O' : '0' + i + 2);
        char c2 = (board[i+2] == 1) ? 'X' : (board[i+2] == 2 ? 'O' : '0' + i + 3);
        std.print_str(" ");
        std.print_char(c0);
        std.print_str(" | ");
        std.print_char(c1);
        std.print_str(" | ");
        std.print_char(c2);
        std.print_str("\n");
        if (i < 6) { std.print_str("---+---+---\n"); }
    }
    std.print_str("\n");
}

check_win = (p: int): int
{
    /* rows */
    for (int i = 0; i < 9; i += 3)
        if (board[i] == p && board[i+1] == p && board[i+2] == p) return 1;
    /* cols */
    for (int i = 0; i < 3; ++i)
        if (board[i] == p && board[i+3] == p && board[i+6] == p) return 1;
    /* diags */
    if (board[0] == p && board[4] == p && board[8] == p) return 1;
    if (board[2] == p && board[4] == p && board[6] == p) return 1;
    return 0;
}

check_draw = (): int
{
    for (int i = 0; i < 9; ++i)
        if (board[i] == 0) return 0;
    return 1;
}

main = () -> int !IO
{
    std.print_str("=== Tic-Tac-Toe ===\n");
    std.print_str("Enter position 1-9:\n");
    
    while (1) {
        print_board();
        std.print_str("Player ");
        std.print_char(current == 1 ? 'X' : 'O');
        std.print_str(" turn (1-9): ");
        
        int pos = std.read_int();  /* reads integer from stdin */
        if (pos < 1 || pos > 9 || board[pos-1] != 0) {
            std.print_str("Invalid move!\n");
            continue;
        }
        board[pos-1] = current;
        
        if (check_win(current)) {
            print_board();
            std.print_str("Player ");
            std.print_char(current == 1 ? 'X' : 'O');
            std.print_str(" wins!\n");
            break;
        }
        if (check_draw()) {
            print_board();
            std.print_str("Draw!\n");
            break;
        }
        current = (current == 1) ? 2 : 1;
    }
    return 0;
}
